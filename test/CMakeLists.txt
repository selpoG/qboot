set(MY_TARGET bootstrap_example)
add_executable(${MY_TARGET} src/main.cpp)
target_link_libraries(${MY_TARGET} PRIVATE qboot)
set_default_property(${MY_TARGET} ..)
set_target_properties(${MY_TARGET} PROPERTIES OUTPUT_NAME test)

set(MY_TARGET comp)
add_executable(${MY_TARGET} src/comp.cpp)
target_link_libraries(${MY_TARGET} PRIVATE qboot)
set_default_property(${MY_TARGET} ..)

add_executable(qboot_smoke src/smoke.cpp)
target_link_libraries(qboot_smoke PRIVATE qboot)
set_default_property(qboot_smoke ..)
add_test(NAME qboot_smoke COMMAND qboot_smoke)
set_tests_properties(qboot_smoke PROPERTIES TIMEOUT 120)

add_executable(qboot_json src/json.cpp)
target_link_libraries(qboot_json PRIVATE qboot)
set_default_property(qboot_json ..)
add_test(NAME qboot_json COMMAND qboot_json)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
	add_test(NAME qboot_json_format COMMAND ${Python3_EXECUTABLE}
		${CMAKE_CURRENT_SOURCE_DIR}/check_json.py $<TARGET_FILE:qboot_json>)
endif()

add_executable(qboot_regression src/regression.cpp)
target_link_libraries(qboot_regression PRIVATE qboot)
set_default_property(qboot_regression ..)
foreach(case scalar subtraction norm linear shift parsing parallel queue)
	add_test(NAME regression_${case} COMMAND qboot_regression)
	set_tests_properties(regression_${case} PROPERTIES
		ENVIRONMENT "QBOOT_REGRESSION=${case}" TIMEOUT 15)
endforeach()
