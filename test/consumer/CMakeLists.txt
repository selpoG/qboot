cmake_minimum_required(VERSION 3.12.4)
project(qboot_consumer LANGUAGES CXX)

find_package(QBoot 0.8.2 REQUIRED)
add_executable(qboot_consumer ../src/smoke.cpp)
target_link_libraries(qboot_consumer PRIVATE QBoot::qboot)
set_target_properties(qboot_consumer PROPERTIES CXX_EXTENSIONS OFF CXX_STANDARD_REQUIRED ON)
if(MSVC)
	target_compile_options(qboot_consumer PRIVATE /W4 /WX)
else()
	target_compile_options(qboot_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME installed_qboot COMMAND qboot_consumer)
set_tests_properties(installed_qboot PROPERTIES TIMEOUT 120)
